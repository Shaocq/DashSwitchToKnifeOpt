# ============================================================
# 奔跑自动切刀（优化版）配置文件
#
# 使用方式：把本文件复制为同目录下的 config.txt，修改后约 1 秒内自动生效。
#
# 如果启用了 ModSetting（工坊 3595729494），也可以直接在游戏的
# 「设置 → Mod 设置」里调整本 mod 的两个选项，无需编辑本文件。
# 注意：ModSetting 里保存/修改过的选项优先于本文件；本文件作为
#       ModSetting 中尚未设置过的选项的默认值。
#
# 如果 mod 目录里没有 config.txt，则回退读取游戏安装目录下的旧版配置：
#   Duckov_Data/StreamingAssets/DashSwitchToKnife_SwitchConfig.txt
#   Duckov_Data/StreamingAssets/DashSwitchToKnife_DisableAttack.txt（文件存在即开启）
# ============================================================

# 战斗输入（无需配置，默认开启）：
#   奔跑中由本 mod 换成近战武器后，按下瞄准（右键）或开火（左键）会自动切回
#   奔跑前的枪械并中断本次奔跑；切回后约 0.15 秒的抬枪时间属正常现象。
#   战斗打断后本 mod 进入休眠：即使仍按着 Shift 自动恢复奔跑，也不会再切刀；
#   真正停下来约 0.4 秒，或重新按下奔跑键之后，下一次起跑才会开启新一轮判定。
#   该功能依赖前置「HarmonyLib」（工坊 3589088839），没有前置时切刀仍正常，
#   只有瞄准/开火自动切回原枪不可用。
#
# 奔跑时切换到哪个装备槽，可选：
#   MeleeWeapon     （近战武器槽，默认）
#   PrimaryWeapon   （主武器槽）
#   SecondaryWeapon （副武器槽）
slot = MeleeWeapon

# 奔跑过程中禁止近战攻击：true / false
# 需要前置「HarmonyLib」（工坊 3589088839）才能生效；
# 前置未启用时切刀功能正常，仅此项不起作用。
disableAttack = false
